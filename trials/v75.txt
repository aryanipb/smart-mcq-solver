import socket, time

def _check(host, port=443, timeout=6):
    t0 = time.time()
    try:
        socket.create_connection((host, port), timeout=timeout).close()
        return True, time.time() - t0
    except Exception as e:
        return False, str(e)

# ── Metric helpers ────────────────────────────────────────────────────────────
def map_at_3(probs_list, labels):
    score = 0.0
    for p, lbl in zip(probs_list, labels):
        for k, idx in enumerate(np.argsort(p)[::-1][:3]):
            if idx == lbl:
                score += 1.0 / (k + 1)
                break
    return score / len(labels)

def top3_str(p):
    return ' '.join([cl[i] for i in np.argsort(p)[::-1][:3]])

def get_probs_batched(model, tok, texts, batch_size=64, aug_prefix=None):
    if aug_prefix:
        texts = [aug_prefix + t for t in texts]
    enc = tok(texts, padding='max_length', truncation=True,
               max_length=192, return_tensors='pt')
    all_probs = []
    model.eval()
    with torch.no_grad():
        for i in range(0, len(texts), batch_size):
            ii = enc['input_ids'][i:i+batch_size].to(device)
            am = enc['attention_mask'][i:i+batch_size].to(device)
            logits = model(input_ids=ii, attention_mask=am).logits
            all_probs.append(torch.softmax(logits, -1).cpu().numpy())
    return np.concatenate(all_probs)

class DownloadTimeout(Exception): pass

def _timeout_handler(signum, frame):
    raise DownloadTimeout("download exceeded time limit")

def load_with_retry(loader_fn, *args, max_attempts=4, timeout_s=180, **kwargs):
    for attempt in range(1, max_attempts + 1):
        signal.signal(signal.SIGALRM, _timeout_handler)
        signal.alarm(timeout_s)
        try:
            print(f'  Attempt {attempt}/{max_attempts} (timeout {timeout_s}s)...')
            result = loader_fn(*args, token=HF_TOKEN or None, **kwargs)
            signal.alarm(0)
            return result
        except Exception as e:
            signal.alarm(0)
            wait = min(30, 5 * attempt)
            print(f'  Failed: {e}')
            print(f'  Retry in {wait}s...')
            time.sleep(wait)
    raise RuntimeError(f'Failed after {max_attempts} attempts')

            continue
        has_weights = any(os.path.isfile(os.path.join(d, f))
                           for f in ('pytorch_model.bin', 'model.safetensors'))
        has_tok = any(os.path.isfile(os.path.join(d, f))
                      for f in ('tokenizer.json', 'spm.model', 'vocab.json'))
        if has_weights and has_tok:
            print(f'  Found local "{hint}" model -> {d}')
            return d
    return None

A lightweight 5-layer transformer encoder built entirely from scratch — no pretrained weights at all.

class ScratchTransformer(nn.Module):
    def __init__(self, vocab_size=50265, embed_dim=512, num_heads=8,
                 ff_dim=1024, num_layers=5, max_len=192, num_labels=5, dropout=0.1):
        super().__init__()
        self.embed     = nn.Embedding(vocab_size, embed_dim, padding_idx=1)
        self.pos_embed = nn.Parameter(torch.randn(1, max_len, embed_dim) * 0.02)
        self.drop      = nn.Dropout(dropout)
        enc_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim, nhead=num_heads, dim_feedforward=ff_dim,
            dropout=dropout, batch_first=True, activation='gelu', norm_first=True,
        )
        self.encoder = nn.TransformerEncoder(enc_layer, num_layers=num_layers)
        self.norm    = nn.LayerNorm(embed_dim)
        self.head    = nn.Sequential(nn.Dropout(dropout), nn.Linear(embed_dim, num_labels))
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.trunc_normal_(m.weight, std=0.02)
                if m.bias is not None: nn.init.zeros_(m.bias)
            elif isinstance(m, nn.Embedding):
                nn.init.trunc_normal_(m.weight, std=0.02)

    def forward(self, input_ids, attention_mask=None, labels=None):
        x = self.embed(input_ids) + self.pos_embed[:, :input_ids.size(1), :]
        x = self.drop(x)
        pad_mask = (attention_mask == 0) if attention_mask is not None else None
        x = self.encoder(x, src_key_padding_mask=pad_mask)
        x = self.norm(x)
        msk = attention_mask.unsqueeze(-1).float() if attention_mask is not None \
              else torch.ones_like(x[:, :, :1])
        x   = (x * msk).sum(1) / msk.sum(1).clamp(min=1)
        logits = self.head(x)
        loss   = F.cross_entropy(logits, labels, label_smoothing=0.1) if labels is not None else None
        return type('Out', (), {'loss': loss, 'logits': logits})()

print('=== Ensemble Weight Optimisation ===')

# Phase 1: Grid search for weighted average
best_map = 0.0
best_w   = np.array([0.05, 0.65, 0.30])

for w1 in np.arange(0.00, 0.25, 0.05):
    for w2 in np.arange(0.35, 0.85, 0.05):
        w3 = round(1.0 - w1 - w2, 4)
        if not (0.05 <= w3 <= 0.60):
            continue
        w  = np.array([w1, w2, w3])
        pe = w[0]*val_probs_m1 + w[1]*val_probs_m2 + w[2]*val_probs_m3
        sc = map_at_3(pe, val_lbls)
        if sc > best_map:
            best_map = sc
            best_w   = w.copy()

print(f'Best val MAP@3 (weighted avg) : {best_map:.4f}')
print(f'Weights -> Scratch={best_w[0]:.2f} | DeBERTa={best_w[1]:.2f} | RoBERTa={best_w[2]:.2f}')

# Phase 2: Stacking with Logistic Regression meta-learner
from sklearn.linear_model import LogisticRegression

# Prepare stacking features (probability distributions from each model)
X_val_stack = np.concatenate([val_probs_m1, val_probs_m2, val_probs_m3], axis=1)
X_test_stack = np.concatenate([test_probs_m1, test_probs_m2, test_probs_m3], axis=1)

# Train meta-learner
meta_learner = LogisticRegression(C=1.0, max_iter=1000, random_state=42)
meta_learner.fit(X_val_stack, val_lbls)

# Get stacking predictions
stacking_probs = meta_learner.predict_proba(X_test_stack)
stacking_map = map_at_3(stacking_probs, val_lbls)

print(f'\nStacking val MAP@3: {stacking_map:.4f}')

# Choose best ensemble method
if stacking_map > best_map:
    print(f'\nUsing STACKING (better than weighted average)')
    test_ens = stacking_probs
    final_map = stacking_map
else:
    print(f'\nUsing WEIGHTED AVERAGE (better than stacking)')
    test_ens = best_w[0]*test_probs_m1 + best_w[1]*test_probs_m2 + best_w[2]*test_probs_m3
    final_map = best_map
